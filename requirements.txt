# =============================================================================
# Les packages nécessaires pour la modélisation
# (les notebooks, l'entraînement, l'interface de test, les tests unitaires)
#
# Pour installer :
#     python -m venv .venv
#     .venv\Scripts\Activate.ps1
#     pip install -r requirements.txt
#
# Les versions sont figées avec "==" et pas laissées libres avec ">=". C'est
# la condition pour qu'un entraînement relancé dans six mois donne exactement
# le même résultat. Une montée de version mineure de scikit-learn ou de
# TensorFlow suffit à décaler les scores de quelques dixièmes de point, et on
# ne saurait plus si l'écart vient du modèle ou de la bibliothèque.
#
# Attention : ce fichier n'est pas celui utilisé par l'API en production.
# L'API tourne sur un serveur Azure gratuit limité à 1 Go de mémoire et
# n'embarque qu'une poignée de packages légers. Voir api/requirements.txt.
#
# Testé avec Python 3.13.1 sous Windows 11.
# =============================================================================


# --- Manipuler les données ---------------------------------------------------
pandas==2.3.3
numpy==2.5.2
# pyarrow sait lire et écrire le format Parquet, celui du fichier de travail
# partagé entre les notebooks (data/processed/tweets_prepares.parquet).
pyarrow==25.0.1


# --- Faire des graphiques ----------------------------------------------------
matplotlib==3.11.1
seaborn==0.13.2


# --- Approche 1 : le modèle classique ----------------------------------------
# scikit-learn fournit TF-IDF, la régression logistique, le découpage
# entraînement/test et le calcul des scores.
scikit-learn==1.9.0


# --- Approche 2 : le modèle avancé sur mesure --------------------------------
# TensorFlow et sa surcouche Keras, pour le réseau de neurones récurrent.
tensorflow==2.21.0
# Gensim sert à entraîner les vecteurs de mots (Word2Vec) sur notre corpus,
# et à télécharger ceux de GloVe déjà entraînés sur 2 milliards de tweets.
gensim==4.4.0


# --- Approche 3 : le modèle BERT ---------------------------------------------
# PyTorch en version processeur uniquement, sans CUDA. La machine n'a pas de
# carte graphique dédiée, et la version GPU pèse 2,5 Go pour rien.
torch==2.13.0
# transformers donne accès aux modèles pré-entraînés comme DistilBERT.
transformers==5.16.1
# torchvision sert normalement aux modèles d'images, et on n'en fait pas ici.
# Il est pourtant obligatoire : MLflow l'importe systématiquement quand il
# enregistre un modèle transformers, même pour du texte. Sans lui, la
# sauvegarde échoue sur un ModuleNotFoundError. La version processeur ne pèse
# que 4 Mo.
torchvision==0.28.0


# --- Suivre les expérimentations et gérer les modèles ------------------------
# MLflow enregistre les réglages, les scores et la durée de chaque
# entraînement, et range les modèles dans un catalogue versionné.
mlflow==3.15.2


# --- Les notebooks -----------------------------------------------------------
jupyter==1.1.1
notebook==7.6.2
ipykernel==7.3.0


# --- L'API, pour la développer et la tester en local -------------------------
fastapi==0.141.1
uvicorn==0.52.4


# --- L'interface de test de l'API --------------------------------------------
streamlit==1.62.0
requests==2.34.2
# python-dotenv lit le fichier .env, ou sont rangés la chaîne de connexion
# Application Insights et l'adresse de l'API. Ce fichier n'est pas versionné :
# le code va dans le dépôt, les secrets restent sur la machine.
#
# Il était déjà installé, mais seulement comme dépendance indirecte d'un autre
# package. Une installation propre à partir de ce fichier aurait pu ne pas
# l'avoir, ou l'avoir dans une version différente. Un package qu'on importe
# doit toujours figurer explicitement ici.
python-dotenv==1.2.3


# --- Le suivi de la performance en production --------------------------------
# Envoie les traces vers Azure Application Insights, qui déclenche l'alerte
# quand trop de prédictions sont invalidées par l'utilisateur.
azure-monitor-opentelemetry==1.8.9

# Ces deux lignes ont l'air redondantes, puisque le package ci-dessus installe
# déjà OpenTelemetry. Elles ne le sont pas, et voici pourquoi.
#
# azure-monitor-opentelemetry déclare avoir besoin de "opentelemetry-sdk~=1.36",
# ce qui accepte n'importe quelle version 1.x à partir de 1.36. Or la classe
# LogData, dont il se sert, a été supprimée du SDK en version 1.44. Pip
# installait donc une combinaison qu'il croyait valable et qui plantait à
# l'import, avec un message parfaitement incompréhensible :
#
#     ImportError: cannot import name 'LogData' from 'opentelemetry.sdk._logs'
#
# Je fige donc explicitement la dernière version d'OpenTelemetry qui fonctionne.
# C'est le genre de piège qu'on ne voit qu'à l'exécution, et c'est exactement
# la raison d'être d'un fichier de dépendances aux versions figées.
opentelemetry-sdk==1.43.0
opentelemetry-api==1.43.0


# --- Les tests unitaires -----------------------------------------------------
pytest==9.1.1
# httpx est le client HTTP utilisé par le client de test de FastAPI.
httpx==0.28.1
